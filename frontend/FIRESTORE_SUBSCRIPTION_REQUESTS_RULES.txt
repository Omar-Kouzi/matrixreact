function signedIn() {
  return request.auth != null;
}

function isAdmin() {
  return signedIn()
    && exists(/databases/$(database)/documents/users/$(request.auth.uid))
    && (
      get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == "admin"
      || get(/databases/$(database)/documents/users/$(request.auth.uid)).data.isAdmin == true
    );
}

match /subscriptionRequests/{requestId} {
  // Admin can list/read all requests; customers can read their own request.
  allow read: if isAdmin()
    || (signedIn() && resource.data.uid == request.auth.uid);

  // Customers can create only a request for their own authenticated account.
  allow create: if signedIn()
    && request.resource.data.uid == request.auth.uid
    && request.resource.data.status == "requested"
    && request.resource.data.messageReceived == false
    && request.resource.data.paymentReceived == false
    && request.resource.data.plan in ["plus", "pro"]
    && request.resource.data.billing in ["monthly", "yearly"]
    && request.resource.data.keys().hasAll([
      "uid", "name", "email", "plan", "planName", "billing", "price",
      "status", "messageReceived", "paymentReceived", "paymentMethod",
      "createdAt", "updatedAt"
    ])
    && request.resource.data.keys().hasOnly([
      "uid", "name", "email", "plan", "planName", "billing", "price",
      "status", "messageReceived", "paymentReceived", "paymentMethod",
      "createdAt", "updatedAt"
    ]);

  // Only admin can update tracking stages, admin notes, payment, or delete requests.
  allow update, delete: if isAdmin();
}
